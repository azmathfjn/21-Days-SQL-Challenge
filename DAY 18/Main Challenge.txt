SELECT 
    patient_id AS identifier,
    name AS full_name,
    'Patient' AS type,
    service
FROM patients
WHERE service IN ('surgery', 'emergency')

UNION ALL

SELECT 
    staff_id AS identifier,
    staff_name AS full_name,
    'Staff' AS type,
    service
FROM staff
WHERE service IN ('surgery', 'emergency')

ORDER BY type, service, full_name;
