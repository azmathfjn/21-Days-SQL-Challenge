-- DAY 21 FINAL CHALLENGE
WITH service_level AS (
    SELECT 
        service,
        SUM(patients_admitted) AS total_admissions,
        SUM(patients_refused) AS total_refusals,
        AVG(patient_satisfaction) AS avg_satisfaction
    FROM services_weekly
    GROUP BY service
),

staff_level AS (
    SELECT 
        st.service,
        COUNT(st.staff_id) AS total_staff,
        AVG(sch.weeks_present) AS avg_weeks_present
    FROM staff st
    LEFT JOIN (
        SELECT 
            staff_id,
            COUNT(week) AS weeks_present
        FROM staff_schedule
        GROUP BY staff_id
    ) sch ON st.staff_id = sch.staff_id
    GROUP BY st.service
),

patient_demo AS (
    SELECT 
        service,
        AVG(age) AS avg_age,
        COUNT(patient_id) AS total_patients
    FROM patients
    GROUP BY service
)

SELECT 
    s.service,
    s.total_admissions,
    s.total_refusals,
    s.avg_satisfaction,
    sl.total_staff,
    sl.avg_weeks_present,
    pd.avg_age,
    pd.total_patients,

    -- Performance Score (Weighted: 60% satisfaction + 40% admission rate)
    ROUND(
        (0.6 * s.avg_satisfaction) +
        (0.4 * (s.total_admissions::decimal / NULLIF(pd.total_patients,0))) 
    , 2) AS performance_score

FROM service_level s
LEFT JOIN staff_level sl ON s.service = sl.service
LEFT JOIN patient_demo pd ON s.service = pd.service
ORDER BY performance_score DESC;
